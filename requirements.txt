# 1. Konfigurasi Kunci API dari GitHub Secrets
GEMINI_KEY = os.environ.get("GEMINI_API_KEY")
SUPABASE_URL = os.environ.get("SUPABASE_URL")
SUPABASE_KEY = os.environ.get("SUPABASE_SERVICE_KEY")

genai.configure(api_key=GEMINI_KEY)
supabase = create_client(SUPABASE_URL, SUPABASE_KEY)
model = genai.GenerativeModel('gemini-1.5-flash')

# 2. Sumber Berita (RSS Feed)
RSS_URLS = [
    "https://www.cnbcindonesia.com/market/rss",
    "https://antara-news.com/rss/ekonomi.xml"
]

def parse_and_process():
    for url in RSS_URLS:
        feed = feedparser.parse(url)
        for entry in feed.entries[:3]: # Ambil 3 berita terbaru
            title = entry.title
            link = entry.link
            
            # Cek apakah berita sudah pernah disimpan di Supabase
            check = supabase.table("berita").select("id").eq("source_url", link).execute()
            if check.data:
                continue # Skip jika berita sudah ada
            
            # Minta Gemini AI merangkum berita
            prompt = f"""
            Kamu adalah asisten keuangan anak-anak/pemula. 
            Rangkum berita berikut dengan bahasa yang sangat sederhana, ceria, dan mudah dipahami:
            Judul: {title}
            
            Berikan output berupa format JSON persis seperti ini (tanpa markdown backtick):
            {{
              "summary": "Penjelasan singkat 1 kalimat utuh",
              "bullets": ["Poin 1", "Poin 2", "Poin 3"],
              "sentiment": "green" atau "red" atau "neutral",
              "category": "Saham" atau "Ekonomi" atau "Dunia"
            }}
            """
            
            try:
                response = model.generate_content(prompt)
                import json
                clean_text = response.text.replace("```json", "").replace("```", "").strip()
                data = json.loads(clean_text)
                
                # Simpan ke Supabase
                supabase.table("berita").insert({
                    "title": title,
                    "summary": data.get("summary", ""),
                    "bullets": data.get("bullets", []),
                    "category": data.get("category", "Umum"),
                    "sentiment": data.get("sentiment", "neutral"),
                    "source_url": link
                }).execute()
                print(f"Berhasil menyimpan: {title}")
            except Exception as e:
                print(f"Gagal memproses {title}: {e}")

if __name__ == "__main__":
    parse_and_process()
